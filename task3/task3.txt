Task 3: Working with Interface Inheritance in C#
Visual Programming Lab Manual · Pages 14–16
1. Aim
To demonstrate interface inheritance using explicit interface implementation in C#.
2. Software requirements
- Microsoft Visual Studio
- C# Console Application
3. Procedure / steps
1. Open Microsoft Visual Studio.
2. Select Create a new project.
3. Choose Console App (C#) and click Next.
4. Enter the project name as InterfaceInheritanceExample.
5. Click Create.
6. Open Program.cs and remove the default code.
7. Copy and paste the program given below.
8. Save the file using Ctrl + S.
9. Run the program using Ctrl + F5.
10. Observe the output displayed in the console.
4. Program
Copy this complete program into Program.cs.

using System;

namespace InterfaceInheritanceExample
{
    // Base interface
    public interface IBaseInterface
    {
        void BaseMethod();
    }

    // Derived interface inheriting from base interface
    public interface IDerivedInterface : IBaseInterface
    {
        void DerivedMethod();
    }

    // Class implementing the derived interface
    public class MyClass : IDerivedInterface
    {
        // Explicit implementation of base interface method
        void IBaseInterface.BaseMethod()
        {
            Console.WriteLine(
                "BaseMethod implementation from IBaseInterface.");
        }

        // Explicit implementation of derived interface method
        void IDerivedInterface.DerivedMethod()
        {
            Console.WriteLine(
                "DerivedMethod implementation from IDerivedInterface.");
        }

        // Access methods through interface references
        public void ExecuteMethods()
        {
            IBaseInterface baseInterface = this;
            IDerivedInterface derivedInterface = this;

            baseInterface.BaseMethod();
            derivedInterface.DerivedMethod();
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            MyClass myClass = new MyClass();
            myClass.ExecuteMethods();
        }
    }
}
